cmake_minimum_required(VERSION 3.25)
project(JerzyMultiFX VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
option(MULTIFX_VST3 "Build VST3" OFF)
option(MULTIFX_RPI "Build standalone audio host" ON)
add_library(multifx_dsp core/Engine.cpp effects/clone.cpp effects/atomic.cpp effects/tape.cpp effects/mx.cpp effects/memory.cpp effects/attack.cpp effects/reel.cpp effects/oil.cpp vendor/reel/TapeEngine.cpp)
target_include_directories(multifx_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 target_compile_options(multifx_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
 add_executable(multifx_tests tests/CoreTests.cpp)
 target_link_libraries(multifx_tests PRIVATE multifx_dsp)
 add_test(NAME DSP_state_and_realtime COMMAND multifx_tests)
endif()
if(MULTIFX_VST3)
 set(VST3_SDK_ROOT "/workspace/.setup/dependencies/vst3sdk" CACHE PATH "Pinned Steinberg SDK")
 set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 if(CMAKE_CROSSCOMPILING)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_BUNDLE_FOR_WINDOWS OFF CACHE BOOL "" FORCE)
 endif()
 add_subdirectory("${VST3_SDK_ROOT}" sdk)
 set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
 set(SDK_ROOT "${VST3_SDK_ROOT}")
 if(MINGW)
  get_target_property(sdk_sources sdk SOURCES)
  list(FILTER sdk_sources EXCLUDE REGEX "/utility/dataexchange\\.cpp$")
  set_property(TARGET sdk PROPERTY SOURCES "${sdk_sources}")
 endif()
 smtg_add_vst3plugin(JerzyMultiFX plugin-vst3/Plugin.cpp plugin-vst3/Editor.cpp)
 target_link_libraries(JerzyMultiFX PRIVATE sdk multifx_dsp)
 target_compile_definitions(JerzyMultiFX PRIVATE NOMINMAX)
 if(WIN32)
  target_link_libraries(JerzyMultiFX PRIVATE user32 gdi32 comdlg32)
 endif()
 if(MINGW)
  foreach(target JerzyMultiFX multifx_tests validator moduleinfotool)
   if(TARGET ${target})
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
   endif()
  endforeach()
  target_link_options(validator PRIVATE -municode)
 endif()
endif()
if(MULTIFX_RPI AND UNIX)
 add_executable(jerzy-multifx rpi/main.cpp)
 target_link_libraries(jerzy-multifx PRIVATE multifx_dsp ${CMAKE_DL_LIBS})
 if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(jerzy-multifx PRIVATE -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS jerzy-multifx RUNTIME DESTINATION bin)
endif()
if(MULTIFX_VST3 AND WIN32 AND BUILD_TESTING)
 add_executable(editor_smoke tests/EditorSmoke.cpp)
 target_include_directories(editor_smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VST3_SDK_ROOT})
 target_link_libraries(editor_smoke PRIVATE sdk multifx_dsp user32 gdi32)
 target_compile_definitions(editor_smoke PRIVATE RELEASE=1 NOMINMAX)
 if(MINGW)
  target_link_options(editor_smoke PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
endif()
